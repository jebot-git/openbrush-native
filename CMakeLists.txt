cmake_minimum_required(VERSION 3.21)
project(openbrush_native VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
find_package(LibArchive REQUIRED)
add_library(brush-core src/document.cpp src/tilt.cpp src/mesh.cpp)
target_include_directories(brush-core PUBLIC src)
target_link_libraries(brush-core PUBLIC LibArchive::LibArchive)
target_compile_options(brush-core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(openbrush-tool src/tool.cpp)
target_link_libraries(openbrush-tool PRIVATE brush-core)
include(CTest)
if(BUILD_TESTING)
  add_executable(brush-core-test tests/core.cpp)
  target_compile_options(brush-core-test PRIVATE -UNDEBUG)
  target_link_libraries(brush-core-test PRIVATE brush-core)
  add_test(NAME brush-core COMMAND brush-core-test ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/upstream-one-stroke.tilt)
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(XR REQUIRED IMPORTED_TARGET openxr)
pkg_check_modules(GRAPHICS REQUIRED IMPORTED_TARGET egl gl)
add_library(brush-render src/render.cpp src/graphics.cpp)
target_link_libraries(brush-render PUBLIC brush-core PkgConfig::GRAPHICS)
add_executable(openbrush-native src/main.cpp src/xr.cpp)
target_link_libraries(openbrush-native PRIVATE brush-render PkgConfig::XR)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME native-render COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/render.py $<TARGET_FILE:openbrush-native> ${CMAKE_CURRENT_BINARY_DIR}/native-demo.ppm)
  add_test(NAME tilt-interoperability COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/interop.py $<TARGET_FILE:openbrush-tool> ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/upstream-one-stroke.tilt)
endif()
